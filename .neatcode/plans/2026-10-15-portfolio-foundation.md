# Portfolio Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the terminal-themed single page with an accurate, data-driven "ink & paper" portfolio (home, about, projects, stub case studies, /terminal) that exports as static files.

**Architecture:** Every fact lives in typed modules under `content/`; small pure helpers in `lib/` read them; Server Component pages under `app/` render them with Tailwind v4 tokens. The only client islands are the theme toggle and `/terminal`. `output: 'export'` builds everything to `out/`.

**Tech Stack:** Next 16.2.4 (App Router, static export), React 19.2.4, TypeScript 5, Tailwind CSS v4, `next/font/google` (Geist, Instrument Serif, JetBrains Mono), `next/og`, vitest 3.2.7, sharp 0.34.5, pnpm 10.

**Spec:** `.neatcode/specs/portfolio-foundation.md`

- Never publish a phone or WhatsApp number anywhere (content, terminal, metadata).

- Dynamic routes: `generateStaticParams` + `export const dynamicParams = false`.
- Next 16: `params` is a `Promise` — always `const { slug } = await props.params`. Type pages with the global `PageProps<'/route'>` (generated by `next typegen`). Do not use `next/image` `priority` (deprecated).
- Before writing any Next-specific file, read the matching guide under `node_modules/next/dist/docs/01-app/` (per `AGENTS.md`).
- Client JavaScript on content pages: theme toggle only. `/terminal` is the only other client component.
- Colour tokens exactly as spec §5; body text pairs ≥ 7:1, all other text pairs ≥ 4.5:1. Minimum text size 13 px (`text-[0.8125rem]`), body 16 px.
- Motion: only hover/focus transitions ≤ 200 ms on colour/opacity/transform; all disabled under `prefers-reduced-motion`.
- Nano Banana output is decorative only (covers, OG background); rendered with `alt=""`. Never generate diagrams, screenshots, metrics visuals, or the portrait.
- Site URL until a custom domain exists: `https://shekhar-fawn.vercel.app`.

## Review Focus

1. **Blocked storage / private mode:** `localStorage` throws → theme follows system preference, no console error, page renders. (Pinned in Task 2 theme-script tests.)
2. **Garbage stored theme** (`localStorage.theme = 'purple'`) → treated as unset, falls back to system preference. (Task 2 tests.)
3. **Hostile terminal input** (`<img src=x onerror=alert(1)>`, 5 000-char strings, mixed case, surrounding spaces) → echoed as plain text, never parsed as HTML; commands match case-insensitively after trimming. (Task 6 tests.)
4. **Sparse project entries** (no metrics, no links, no cover, no period) → card and case-study page omit those blocks entirely, no empty rows or "undefined". (Task 3 and Task 4 render tests.)
5. **Unknown project URL** (`/projects/does-not-exist`) → static `404.html` exists and the not-found page links home; build rejects slugs outside the registry. (Task 4 tests + Task 9 build-output test.)

```ts
  end?: string;

  value: string;
  label: string;

}

  slug: string;
```

```ts
import { projects } from "@/content/projects";
import type { Period, Project, ProjectKind, ProjectStatus } from "@/content/types";

const KIND_LABEL: Record<ProjectKind, string> = {
  "open-source": "Open source",
  work: "Work",
  side: "Side project",
};

const STATUS_LABEL: Record<ProjectStatus, string> = {
  shipped: "Shipped",
  active: "Active",
  building: "Building",
  archived: "Archived",
};

const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];

export function getProjects(): Project[] {
  return [...projects].sort((a, b) => a.order - b.order);
}

export function getProject(slug: string): Project | undefined {
  return projects.find((p) => p.slug === slug);
}

export function getFeatured(): Project {
  const featured = projects.find((p) => p.featured);
  if (!featured) throw new Error("content/projects.ts must mark one project as featured");
  return featured;
}
```

```tsx
    const current =
      root.dataset.theme ??
      (window.matchMedia("(prefers-color-scheme: dark)").matches ? "dark" : "light");
```

```js
#!/usr/bin/env node
// Usage: node scripts/optimize-image.mjs <input> <output.webp> <width> [height] [quality=75]
import { mkdirSync } from "node:fs";
import { dirname } from "node:path";
import sharp from "sharp";

const [input, output, width, height, quality = "75"] = process.argv.slice(2);
if (!input || !output || !width) {
  console.error("Usage: node scripts/optimize-image.mjs <input> <output.webp> <width> [height] [quality]");
  process.exit(1);
}

mkdirSync(dirname(output), { recursive: true });
const info = await sharp(input)
  .resize({ width: Number(width), height: height ? Number(height) : undefined, fit: "cover" })
  .webp({ quality: Number(quality) })
  .toFile(output);
console.log(`${output}: ${info.width}x${info.height}, ${(info.size / 1024).toFixed(1)} KB`);
```

```ts
import { skills } from "@/content/about";
import { site } from "@/content/site";
import { getProjects } from "@/lib/projects";

export type TerminalLine = { kind: "input" | "output" | "error"; text: string };
export type CommandResult = { lines: TerminalLine[]; clear?: boolean; open?: string };

const MAX_ECHO = 200;

const COMMANDS: Record<string, { help: string; run: () => string | CommandResult }> = {
  help: { help: "list commands", run: () => helpText() },
  about: {
    help: "who is shekhar",
    run: () => `${site.name} · ${site.role} · ${site.focus}\n${site.headline}\nBased in ${site.location.city}, ${site.location.country}.`,
  },
  skills: {
    help: "what he builds with",
    run: () => skills.map((g) => `${g.name.padEnd(10)}: ${g.items.join(", ")}`).join("\n"),
  },
  projects: {
    help: "shipped work",
    run: () =>
      getProjects()
        .map((p, i) => `[${String(i + 1).padStart(2, "0")}] ${p.title} — ${p.tagline}`)
        .join("\n"),
  },
  contact: {
    help: "reach out",
    run: () => `email    : ${site.email}\ngithub   : ${site.links.github}\nlinkedin : ${site.links.linkedin}`,
  },
  resume: {
    help: "open the résumé",
    run: () => ({ lines: [{ kind: "output", text: "opening resume.pdf…" }], open: site.resume }),
  },
  clear: { help: "clear the screen", run: () => ({ lines: [], clear: true }) },
};

const EASTER_EGGS: Record<string, string> = {
  sudo: "shekhar is not in the sudoers file. this incident will be reported. (just kidding.)",
  whoami: "visitor · thanks for stopping by.",
  coffee: "brewing… done. stay caffeinated.",
};

export const COMMAND_NAMES = Object.keys(COMMANDS);

export const INITIAL_LINES: TerminalLine[] = [
  { kind: "output", text: "Welcome. Type 'help' to see commands." },
];

function helpText(): string {
  return COMMAND_NAMES.map((name) => `${name.padEnd(10)} ${COMMANDS[name].help}`).join("\n");
}

export function runCommand(raw: string): CommandResult {
  const input = raw.trim();
  if (!input) return { lines: [] };

  const echo: TerminalLine = {
    kind: "input",
    text: input.length > MAX_ECHO ? `${input.slice(0, MAX_ECHO)}…` : input,
  };
  const name = input.toLowerCase().split(/\s+/)[0];

  const command = COMMANDS[name];
  if (command) {
    const result = command.run();
    if (typeof result === "string") return { lines: [echo, { kind: "output", text: result }] };
    if (result.clear) return result;
    return { ...result, lines: [echo, ...result.lines] };
  }

  const egg = EASTER_EGGS[name];
  if (egg) return { lines: [echo, { kind: "output", text: egg }] };

  return {
    lines: [echo, { kind: "error", text: `command not found: ${echo.text} — type 'help'` }],
  };
}
```

```tsx
"use client";

import { type FormEvent, type KeyboardEvent, useEffect, useRef, useState } from "react";
import { INITIAL_LINES, runCommand, type TerminalLine } from "./commands";

const LINE_CLASS: Record<TerminalLine["kind"], string> = {
  input: "text-fg",
  output: "text-fg-muted",
  error: "text-accent",
};

export function Terminal() {
  const [lines, setLines] = useState<TerminalLine[]>(INITIAL_LINES);
  const [value, setValue] = useState("");
  const history = useRef<string[]>([]);
  const historyIndex = useRef(-1);
  const logRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    logRef.current?.scrollTo({ top: logRef.current.scrollHeight });
  }, [lines]);

  function submit(e: FormEvent) {
    e.preventDefault();
    const result = runCommand(value);
    if (value.trim()) history.current.unshift(value.trim());
    historyIndex.current = -1;
    setValue("");
    if (result.clear) return setLines([]);
    setLines((prev) => [...prev, ...result.lines]);
    if (result.open) window.open(result.open, "_blank", "noopener");
  }

  function onKeyDown(e: KeyboardEvent<HTMLInputElement>) {
    const h = history.current;
    if (e.key === "ArrowUp" && h.length > 0) {
      e.preventDefault();
      historyIndex.current = Math.min(historyIndex.current + 1, h.length - 1);
      setValue(h[historyIndex.current]);
    } else if (e.key === "ArrowDown") {
      e.preventDefault();
      historyIndex.current = Math.max(historyIndex.current - 1, -1);
      setValue(historyIndex.current === -1 ? "" : h[historyIndex.current]);
    }
  }

  return (
    <div className="overflow-hidden rounded-xl border border-border bg-surface font-mono text-sm">
      <div
        ref={logRef}
        role="log"
        aria-label="Terminal output"
        className="max-h-[60vh] min-h-64 space-y-1 overflow-y-auto whitespace-pre-wrap break-words p-4"
      >
        {lines.map((line, i) => (
          <p key={i} className={LINE_CLASS[line.kind]}>
            {line.kind === "input" && <span aria-hidden="true" className="text-accent">$ </span>}
            {line.text}
          </p>
        ))}
      </div>
      <form
        onSubmit={submit}
        className="flex items-center gap-2 border-t border-border px-4 py-3 focus-within:outline-2 focus-within:outline-accent"
      >
        <label htmlFor="terminal-input" className="text-accent">
          <span aria-hidden="true">$</span>
          <span className="sr-only">Command</span>
        </label>
        <input
          id="terminal-input"
          value={value}
          onChange={(e) => setValue(e.target.value)}
          onKeyDown={onKeyDown}
          autoComplete="off"
          spellCheck={false}
          placeholder="type a command and press enter"
          className="min-w-0 flex-1 bg-transparent text-fg placeholder:text-fg-meta focus-visible:outline-none"
        />
      </form>
    </div>
  );
}
```

```ts
import { existsSync, readdirSync, readFileSync } from "node:fs";
import { join } from "node:path";
import { describe, expect, it } from "vitest";
import { projects } from "../content/projects";

const OUT = join(process.cwd(), "out");
const read = (f: string) => readFileSync(join(OUT, f), "utf8");
const hasPrefix = (dir: string, prefix: string) =>
  existsSync(join(OUT, dir)) && readdirSync(join(OUT, dir)).some((f) => f.startsWith(prefix));

const PAGES = [
  "index.html",
  "about.html",
  "projects.html",
  "terminal.html",
  "404.html",
  ...projects.map((p) => `projects/${p.slug}.html`),
];

describe("static export", () => {
  it.each(PAGES)("emits %s", (f) => {
    expect(existsSync(join(OUT, f))).toBe(true);
  });

  it.each(["sitemap.xml", "robots.txt", "resume.pdf", "about/portrait.webp", "icon.svg", "apple-icon.png"])(
    "emits %s",
    (f) => expect(existsSync(join(OUT, f))).toBe(true),
  );

  it("emits OG images for the site and each project", () => {
    expect(hasPrefix(".", "opengraph-image") || existsSync(join(OUT, "og"))).toBe(true);
    for (const p of projects) {
      expect(hasPrefix(`projects/${p.slug}`, "opengraph-image") || existsSync(join(OUT, "og", `${p.slug}.png`))).toBe(true);
    }
  });

  it.each(PAGES)("%s has lang=en, exactly one h1, and no phone number", (f) => {
    const html = read(f);
    expect(html).toContain('<html lang="en"');
    expect(html.match(/<h1[\s>]/g)).toHaveLength(1);
    expect(html).not.toMatch(/8409209072|84092\s?09072/);
  });

  it("marks /terminal noindex and keeps it out of the sitemap", () => {
    expect(read("terminal.html")).toMatch(/<meta name="robots" content="noindex/);
    expect(read("sitemap.xml")).not.toContain("/terminal");
  });

  it("ships the JSON-LD person on the home page", () => {
    expect(read("index.html")).toContain('"@type":"Person"');
  });

  it("contains none of the dropped claims", () => {
    const all = PAGES.map(read).join("\n");
    for (const banned of ["99.9%", "420ms", "fine-tuned models", "6000+", "CLEARANCE", "no frameworks"]) {
      expect(all).not.toContain(banned);
    }
  });
});
```
